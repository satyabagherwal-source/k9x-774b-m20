def validate_actions(hook):
    """Validate saved JSON against the installed plugins and config schemas."""
    from archivebox.core.models import Folder

    if not isinstance(hook.actions, list) or len(hook.actions) > 100:
        raise ValidationError({"actions": "Use an ordered list of at most 100 actions."})
    for index, action in enumerate(hook.actions, 1):
        try:
            if not isinstance(action, dict) or action.get("type") not in ACTION_FIELDS:
                raise ValueError("Choose a supported action.")
            kind = action["type"]
            if action.keys() - (ACTION_FIELDS[kind] | {"type"}):
                raise ValueError("Unexpected action fields.")
            snapshot = hook.ref == "archivebox.core.models.Snapshot"
            crawl = hook.ref == "archivebox.crawls.models.Crawl"
            if kind in CAPTURE_ACTIONS and not (snapshot or crawl):
                raise ValueError("Capture settings apply to Snapshot or Crawl.")
            if kind in SNAPSHOT_ACTIONS and not snapshot:
                raise ValueError("This action requires a Snapshot.")
            if kind in CAPTURE_ACTIONS | SNAPSHOT_ACTIONS and hook.signal == "DELETE":
                raise ValueError("A deleted object cannot be modified.")
            if kind in {"enable_plugins", "disable_plugins"}:
                names = action.get(kind)
                catalog = get_plugin_catalog()
                if not isinstance(names, list) or not names or any(not isinstance(name, str) or name not in catalog for name in names):
                    raise ValueError("Select installed plugins by their exact name.")
            elif kind == "config_overrides":
                overrides = action.get(kind)
                if not isinstance(overrides, dict) or not overrides:
                    raise ValueError("Choose at least one configuration setting.")
                invalid = overrides.keys() - configurable_fields().keys()
                if invalid:
                    raise ValueError(f"Unknown or non-capture settings: {', '.join(sorted(invalid))}.")
                resolved = get_config(overrides=overrides, include_machine=False).model_dump(mode="json")
                action[kind] = {key: resolved[key] for key in overrides}
                if "PLUGINS" in overrides:
                    if any(name not in get_plugin_catalog() for name in (parse_plugin_selection(overrides["PLUGINS"]) or ())):
                        raise ValueError("PLUGINS must contain installed plugin names.")
            elif kind == "max_depth":
                depth = action.get(kind)
                if not crawl or not isinstance(depth, int) or isinstance(depth, bool) or not 0 <= depth <= 4:
                    raise ValueError("Choose a Crawl depth from 0 to 4.")
            elif kind in {"add_tags", "remove_tags"}:
                value = action.get(kind)
                if not isinstance(value, str) or not value.strip() or any(len(name.strip()) > 100 for name in value.split(",")):
                    raise ValueError("Enter comma-separated tags, at most 100 characters each.")
            elif kind == "permissions" and action.get(kind) not in {"private", "unlisted", "public"}:
                raise ValueError("Choose private, unlisted, or public.")
            elif kind == "folder" and action.get("folder_id"):
                if not Folder.objects.destinations().filter(pk=action["folder_id"]).exists():
                    raise ValueError("Choose a regular folder or a folder with children.")
            elif kind in {"python", "shell"}:
                value = action.get("code" if kind == "python" else "command")
                if not isinstance(value, str) or not value.strip():
                    raise ValueError("Enter code to execute.")
                if kind == "python":
                    compile(value, "<rule>", "exec")
            elif kind == "http_request":
                endpoint = action.get("endpoint", "")
                if not isinstance(endpoint, str) or not endpoint:
                    raise ValueError("Enter a request URL.")
                relative = endpoint.startswith("/api/")
                if not relative and (urlsplit(endpoint).scheme not in {"http", "https"} or not urlsplit(endpoint).netloc):
                    raise ValueError("Use an HTTP(S) URL or an ArchiveBox /api/ path.")

def _render(value, payload):
    """Substitute event fields only; scripts receive JSON instead of interpolation."""
    if isinstance(value, dict):
        return {key: _render(item, payload) for key, item in value.items()}
    if isinstance(value, list):
        return [_render(item, payload) for item in value]
    if not isinstance(value, str):
        return value

    def resolve(match):
        path = match.group(1).split(".")
        current = payload
        if path[0] in {"object", "event"} and len(path) > 1:
            path = path[1:]
        for key in path:
            if not isinstance(current, dict) or key not in current:
                raise ValueError(f"Unknown event field: {match.group(1)}")
            current = current[key]
        return current

    pattern = r"{{\s*([A-Za-z_][\w.]*)\s*}}"
    if match := re.fullmatch(pattern, value):
        return resolve(match)
    return re.sub(pattern, lambda match: str(resolve(match)), value)

        finally:
            if token is not None:
                executing_webhook.reset(token)

from django.apps import apps

class MembershipChange:
    def __init__(self, snapshot_ids=None, *, deleted=None):
        from archivebox.api.webhook_events import executing_webhook

        if executing_webhook.get():
            self.folder_ids = []
            return
        from archivebox.api.models import Rule
        from archivebox.core.models import Snapshot
        from archivebox.core.smart_folders import RuleQuery

        scopes = list(
            Rule.objects.filter(
                Q(folder_id__isnull=False) | Q(any_folder=True),
                enabled=True,
                ref="archivebox.core.models.Snapshot",
                signal__in=["FOLDER_ENTERED", "FOLDER_LEFT"],
            )
            .order_by()
            .values_list("folder_id", "any_folder")
            .distinct(),
        )
        self.folder_ids = [str(folder_id) for folder_id, _ in scopes if folder_id]
        self.any_folder = any(any_folder for _, any_folder in scopes)
        self.deleted = deleted or {}
        self.before = {}
        if not self.folder_ids and not self.any_folder:
            return
        self.ids = None if snapshot_ids is None else set(snapshot_ids() if callable(snapshot_ids) else snapshot_ids)
        self.compiler = RuleQuery(freeze_limits=True)
        if self.any_folder:
            self.folder_ids = list(self.compiler.folders)
        self.queries = {folder_id: self.compiler.folder(folder_id) for folder_id in self.folder_ids}
        candidates = self.ids
        for folder_id, query in self.queries.items():
            matches = Snapshot.objects.filter(query)
            if candidates is not None:
                matches = matches.filter(pk__in=candidates)
            self.before[folder_id] = set(matches.values_list("pk", flat=True))
        # Only a limited folder can change an untouched row's membership when
        # snapshot data changes. Follow the compiler's cache to include limits in
        # nested/referenced folders without inventing a second dependency walker.
        self.has_limits = any((self.compiler.folders[key].rules or {}).get("limit") for key in self.compiler.cache)

    def finish(self):
        if not self.folder_ids:
            return
        from archivebox.core.models import Snapshot
        from archivebox.core.smart_folders import RuleQuery
        from archivebox.api.webhook_events import emit_event

        after_compiler = RuleQuery(freeze_limits=True)
        after_queries = {folder_id: after_compiler.folder(folder_id) for folder_id in self.folder_ids}
        events = []
        for folder_id, query in after_queries.items():
            matches = Snapshot.objects.filter(query)
            if self.ids is not None:
                matches = matches.filter(pk__in=self.ids)
            after = set(matches.values_list("pk", flat=True))
            before = self.before[folder_id]
            left, entered = before - after, after - before
            if self.ids is not None and self.has_limits:
                # Compare frozen ordered windows in SQL for unchanged snapshots.
                # This finds evictions/refills without materializing a million-row
                # playlist or putting its IDs into a giant SQL IN expression.
                untouched = Snapshot.objects.exclude(pk__in=self.ids)
                old_query = self.queries[folder_id]
                old_ids = untouched.filter(old_query).order_by().values_list("pk", flat=True)
                new_ids = untouched.filter(query).order_by().values_list("pk", flat=True)
                left.update(old_ids.difference(new_ids))
                entered.update(new_ids.difference(old_ids))
            for signal, ids in (("FOLDER_LEFT", left), ("FOLDER_ENTERED", entered)):
                # Finish all SELECTs before dispatch: webhook actions can perform
                # subprocess/network work and must not retain a database cursor.
                instances = {snapshot.pk: snapshot for snapshot in Snapshot.objects.filter(pk__in=ids)}
                for pk in sorted(ids):
                    instance = instances.get(pk) or self.deleted.get(pk)
                    if instance is not None:
                        events.append((instance, signal, str(folder_id)))
        # An action may itself edit tags/folders. Capture the complete transition
        # first, so earlier subscriptions cannot alter later membership deltas.
        for instance, signal, folder_id in events:
            emit_event(instance, signal, folder_id=folder_id)

def submitted_rule(request):
    payload = json.loads(request.body)
    if not isinstance(payload, dict):
        raise ValidationError("Expected a rule object.")
    allowed = {
        "id",
        "name",
        "ref",
        "signal",
        "enabled",
        "folder_id",
        "tag_id",
        "relationship",
        "actions",
        "stop_processing",
        "stop_on_failure",
        "auth_token",
        "clear_auth_token",
        "keep_last_response",
    }
    if payload.keys() - allowed:
        raise ValidationError("Unknown rule fields.")
    for field in ("enabled", "stop_processing", "stop_on_failure", "clear_auth_token", "keep_last_response"):
        if field in payload and not isinstance(payload[field], bool):
            raise ValidationError(f"{field} must be a boolean.")
    if "auth_token" in payload and not isinstance(payload["auth_token"], str):
        raise ValidationError("auth_token must be a string.")
    hook = get_object_or_404(Rule, pk=payload["id"]) if payload.get("id") else Rule(created_by=request.user)
    previous_signal = hook.signal
    for field in allowed - {"id", "auth_token", "clear_auth_token", "tag_id", "relationship"}:
        if field in payload:
            value = payload[field]
            if field == "folder_id":
                hook.any_folder = value == "*"
                value = None if hook.any_folder else value
            setattr(hook, field, value)
    if "relationship" in payload:
        value = payload["relationship"]
        if value is not None and not isinstance(value, str):
            raise ValidationError("Relationship must be a string.")
        if hook.signal not in TAG_EVENTS:
            hook.relationship = "" if value in (None, "*") else value
        elif value not in (None, ""):
            raise ValidationError("Choose a tag for a tag event.")
    if hook.signal != previous_signal and (hook.signal in TAG_EVENTS) != (previous_signal in TAG_EVENTS):
        if hook.signal in TAG_EVENTS or "relationship" not in payload:
            hook.relationship = ""
    if "tag_id" in payload:
        value = payload["tag_id"]
        if value is not None and (hook.signal not in TAG_EVENTS or (value != "*" and (type(value) is not int or value <= 0))):
            raise ValidationError("Choose an existing tag or Any tag for a tag event.")
        if hook.signal in TAG_EVENTS:
            hook.relationship = "" if value is None else str(value)
    if payload.get("relationship") == "*" and hook.signal not in RELATIONSHIP_EVENTS:
        raise ValidationError("Choose a relationship for a relationship event only.")
    if payload.get("clear_auth_token"):
        hook.auth_token = ""
    elif payload.get("auth_token"):
        hook.auth_token = payload["auth_token"]
    hook.full_clean()
    return hook


def validation_error(error):
    return JsonResponse({"error": "; ".join(error.messages) if isinstance(error, ValidationError) else "Invalid rule data."}, status=400)
