function toCamelCase(str) {
    // 将下划线和中划线统一替换为空格
    let camelCaseStr = str.replace(/[-_]/g, ' ');

    // 将每个单词的首字母大写，其余字母小写
    camelCaseStr = camelCaseStr.split(' ').map(word => {
        return word.charAt(0).toUpperCase() + word.slice(1).toLowerCase();
    }).join('');

    return camelCaseStr;
}

// 读取所有的icon
