/// The forms a source may take in the file: as recorded, or with const and let swapped (a Node
/// port switches its function's const to let while enabled).
fn forms(source: &str) -> [String; 3] {
    let source = source.strip_prefix('\u{feff}').unwrap_or(source).replace("\r\n", "\n");
    [source.replacen("const ", "let ", 1), source.replacen("let ", "const ", 1), source]
}

/// The file without fast-anything's own marked blocks: an enabled Java, Go, C#, C/C++, PHP or
/// Kotlin port puts one inside the function's body, and that isn't a change to the original.
fn without_blocks(text: &str) -> String {
    let text = without_inline(text.strip_prefix('\u{feff}').unwrap_or(text));
    let mut out = String::with_capacity(text.len());
    let mut inside = false;
    for line in text.split_inclusive('\n') {
        if !inside && line.contains(">>> fast-anything: ") {
            inside = true;
        }
        if !inside {
            out.push_str(line);
        }
        if inside && line.contains("<<< fast-anything: ") {
            inside = false;
        }
    }
    out
}

/// The text without the blocks fast-anything puts inside a line (` /* // >>> fast-anything: id >>> */ ...
/// /* // <<< fast-anything: id <<< */`, after a Kotlin expression body's `=`).
fn without_inline(text: &str) -> String {
    let (begin, end) = (" /* // >>> fast-anything: ", " <<< */");
    let mut out = String::with_capacity(text.len());
    let mut rest = text;
    while let Some(b) = rest.find(begin) {
        let line_end = rest[b..].find('\n').map_or(rest.len(), |n| b + n);
        let close = rest[b..line_end]
            .find("/* // <<< fast-anything: ")
            .and_then(|c| rest[b + c..line_end].find(end).map(|e| b + c + e + end.len()));
        let Some(close) = close else { break };
        out.push_str(&rest[..b]);
        rest = &rest[close..];
    }
    out.push_str(rest);
    out
}

/// The source appears in the file as whole lines.
pub(crate) fn found(text: &str, source: &str) -> bool {
    let text = format!("\n{}\n", without_blocks(text).replace("\r\n", "\n"));
    forms(source).iter().any(|s| text.contains(&format!("\n{}\n", s.trim_end_matches('\n'))))
}

/// The sources that start a line in the file, each extended to the end of its last line so that
/// found() can match it as whole lines (see Port::sources).
pub fn verbatim(text: &str, sources: &[&str]) -> Vec<String> {
    let text = format!("\n{}\n", without_blocks(text).replace("\r\n", "\n"));
    sources
        .iter()
        .filter(|s| !s.trim().is_empty())
        .filter_map(|s| {
            forms(s).iter().find_map(|form| {
                let at = text.find(&format!("\n{form}"))? + 1;
                let end = at + form.len() + text[at + form.len()..].find('\n')?;
                Some(text[at..end].to_string())
            })
        })
        .collect()
}

use anyhow::{Context, Result};
use std::io::Read;
use std::path::{Component, Path, PathBuf};
use std::process::{Command, Output, Stdio};

pub fn absolute(p: &Path) -> PathBuf {
    std::path::absolute(p).unwrap_or_else(|_| p.to_path_buf())
}

/// Path from directory from to to, both absolute.
pub fn relative(from: &Path, to: &Path) -> PathBuf {
    let a: Vec<Component> = from.components().collect();
    let b: Vec<Component> = to.components().collect();
    let common = a.iter().zip(&b).take_while(|(x, y)| x == y).count();
    let mut out = PathBuf::new();
    for _ in common..a.len() {
        out.push("..");
    }
    for c in &b[common..] {
        out.push(c);
    }
    out
}

/// Like [`output`], but stops the command once `limit` has passed: asks it to close first (WM_CLOSE on Windows,
/// SIGTERM elsewhere), so it can shut down cleanly and write what it has, then ends it for good after `grace`.
/// Also says whether it had to be stopped.
pub fn output_for(cmd: &mut Command, what: &str, limit: Option<Duration>, grace: Duration) -> Result<(Output, bool)> {
    output_signalling(cmd, what, limit, grace, "TERM")
}

/// output_for, asking the process to stop with this signal on Unix (INT is Ctrl+C).
pub fn output_signalling(
    cmd: &mut Command,
    what: &str,
    limit: Option<Duration>,
    grace: Duration,
    signal: &str,
) -> Result<(Output, bool)> {
    output_stopping(cmd, what, limit, grace, signal, || true)
}

/// output_signalling, for an app that saves its own results on time: where it can't be asked to close (a
/// console app on Windows), it's left to run, at most `grace` longer, until `saved` says they're written.
pub fn output_stopping(
    cmd: &mut Command,
    what: &str,
    limit: Option<Duration>,
    grace: Duration,
    signal: &str,
    saved: impl Fn() -> bool,
) -> Result<(Output, bool)> {
    let Some(limit) = limit else { return Ok((output(cmd, what)?, false)) };
    let mut child = cmd
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .with_context(|| format!("couldn't start {what} ({})", cmd.get_program().to_string_lossy()))?;
    fn drain<R: Read + Send + 'static>(r: Option<R>) -> std::thread::JoinHandle<Vec<u8>> {
        std::thread::spawn(move || {
            let mut buf = Vec::new();
            if let Some(mut r) = r {
                let _ = r.read_to_end(&mut buf);
            }
            buf
        })
    }
    let (stdout, stderr) = (drain(child.stdout.take()), drain(child.stderr.take()));
    let started = Instant::now();
    let mut asked: Option<(Instant, bool)> = None;
    let status = loop {
        if let Some(status) = child.try_wait()? {
            break status;
        }
        match asked {
            None if started.elapsed() >= limit => asked = Some((Instant::now(), ask_to_close(child.id(), signal))),
            Some((at, closing)) if at.elapsed() >= grace || (!closing && saved()) => {
                let _ = child.kill();
                break child.wait()?;
            }
            _ => {}
        }
        std::thread::sleep(Duration::from_millis(50));
    };
    let out = Output { status, stdout: stdout.join().unwrap_or_default(), stderr: stderr.join().unwrap_or_default() };
    Ok((out, asked.is_some()))
}

/// Ask a process to close the way a user would. False if it can't be asked (a console app on Windows).
fn ask_to_close(pid: u32, signal: &str) -> bool {
    let flag = format!("-{signal}");
    let (program, args) = if cfg!(windows) { ("taskkill", ["/PID"]) } else { ("kill", [flag.as_str()]) };
    Command::new(program)
        .args(args)
        .arg(pid.to_string())
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .is_ok_and(|s| s.success())
}

pub fn thousands(n: u64) -> String {
    let s = n.to_string();
    let mut out = String::new();
    for (i, c) in s.chars().enumerate() {
        if i > 0 && (s.len() - i).is_multiple_of(3) {
            out.push(',');
        }
        out.push(c);
    }
    out
}

// Fuzzy string matching helpers.
#include <algorithm>
#include <cctype>
#include <string>
#include <utility>
#include <vector>

// Edit distance (by byte): how many inserts, deletes or substitutions turn a into b.
static int levenshtein(const std::string &a, const std::string &b) {
    const std::string &s = a.size() >= b.size() ? a : b;
    const std::string &t = a.size() >= b.size() ? b : a;
    std::vector<int> previous(t.size() + 1), current(t.size() + 1);
    for (size_t j = 0; j <= t.size(); j++) previous[j] = (int) j;
    for (size_t i = 1; i <= s.size(); i++) {
        current[0] = (int) i;
        for (size_t j = 1; j <= t.size(); j++) {
            int cost = s[i - 1] == t[j - 1] ? 0 : 1;
            current[j] = std::min({previous[j] + 1, current[j - 1] + 1, previous[j - 1] + cost});
        }
        std::swap(previous, current);
    }
    return previous[t.size()];
}

// Lowercase (ASCII) and squeeze whitespace.
static std::string normalize(const std::string &name) {
    std::string out;
    bool space = false;
    for (unsigned char c : name) {
        if (std::isspace(c)) {
            space = !out.empty();
            continue;
        }
        if (space) out += ' ';
        space = false;
        out += (char) std::tolower(c);
    }
    return out;
}

// Index pairs of names within max_distance edits of each other.
std::vector<std::pair<int, int>> find_duplicates(const std::vector<std::string> &names, int max_distance) {
    std::vector<std::string> clean;
    for (const auto &n : names) clean.push_back(normalize(n));
    std::vector<std::pair<int, int>> pairs;
    for (size_t i = 0; i < clean.size(); i++) {
        for (size_t j = i + 1; j < clean.size(); j++) {
            if (levenshtein(clean[i], clean[j]) <= max_distance) pairs.emplace_back((int) i, (int) j);
        }
    }
    return pairs;
}

/// The C++ normalize: ASCII lowercase, whitespace (std::isspace in the C locale) squeezed and trimmed, as bytes.
fn normalize(name: &str) -> Vec<u8> {
    let mut out = Vec::with_capacity(name.len());
    let mut space = false;
    for &c in name.as_bytes() {
        if matches!(c, b' ' | b'\t' | b'\n' | b'\x0B' | b'\x0C' | b'\r') {
            space = !out.is_empty();
            continue;
        }
        if space {
            out.push(b' ');
        }
        space = false;
        out.push(c.to_ascii_lowercase());

/// Edit distance, one row on the shorter side, common prefix and suffix skipped.
fn distance<T: PartialEq>(a: &[T], b: &[T], row: &mut Vec<usize>) -> usize {
    let prefix = a.iter().zip(b).take_while(|(x, y)| x == y).count();
    let (a, b) = (&a[prefix..], &b[prefix..]);
    let suffix = a.iter().rev().zip(b.iter().rev()).take_while(|(x, y)| x == y).count();
    let (a, b) = (&a[..a.len() - suffix], &b[..b.len() - suffix]);
    let (long, short) = if a.len() >= b.len() { (a, b) } else { (b, a) };
    if short.is_empty() {
        return long.len();
    }
    row.clear();
    row.extend(0..=short.len());
    for (i, ca) in long.iter().enumerate() {
        let mut diagonal = row[0];
        row[0] = i + 1;
        for (j, cb) in short.iter().enumerate() {
            let above = row[j + 1];
            row[j + 1] = (above + 1).min(row[j] + 1).min(diagonal + usize::from(ca != cb));
            diagonal = above;
        }
    }
    row[short.len()]
}

/// All (i, j) with i < j and distance <= max, in the original loop order, spread over threads.
fn pairs<T: PartialEq + Sync>(clean: &[Vec<T>], max: usize) -> Vec<(usize, usize)> {
    let threads = std::thread::available_parallelism().map_or(1, |n| n.get()).min(clean.len().max(1));
    let mut found: Vec<(usize, usize)> = std::thread::scope(|s| {
        let workers: Vec<_> = (0..threads)
            .map(|t| {
                s.spawn(move || {
                    let mut row = Vec::new();
                    let mut out = Vec::new();
                    for i in (t..clean.len()).step_by(threads) {
                        for j in i + 1..clean.len() {
                            if clean[i].len().abs_diff(clean[j].len()) <= max && distance(&clean[i], &clean[j], &mut row) <= max {
                                out.push((i, j));
                            }
                        }
                    }
                    out
                })
            })
            .collect();
        workers.into_iter().flat_map(|w| w.join().expect("worker panicked")).collect()
    });

package main

import "strings"

// levenshtein is the edit distance: how many inserts, deletes or substitutions turn a into b.
func levenshtein(a, b string) int {
	ra, rb := []rune(a), []rune(b)
	if len(ra) < len(rb) {
		ra, rb = rb, ra
	}
	previous := make([]int, len(rb)+1)
	for j := range previous {
		previous[j] = j
	}
	for i := 1; i <= len(ra); i++ {
		current := make([]int, len(rb)+1)
		current[0] = i
		for j := 1; j <= len(rb); j++ {
			cost := 1
			if ra[i-1] == rb[j-1] {
				cost = 0
			}
			current[j] = min(previous[j]+1, current[j-1]+1, previous[j-1]+cost)
		}
		previous = current
	}
	return previous[len(rb)]
}

func normalize(name string) string {
	return strings.Join(strings.Fields(strings.ToLower(name)), " ")
}

// findDuplicates returns index pairs of names within maxDistance edits of each other.
func findDuplicates(names []string, maxDistance int) [][2]int {
	clean := make([]string, len(names))
	for i, n := range names {
		clean[i] = normalize(n)
	}
	var pairs [][2]int
	for i := range clean {
		for j := i + 1; j < len(clean); j++ {
			if levenshtein(clean[i], clean[j]) <= maxDistance {
				pairs = append(pairs, [2]int{i, j})
			}
		}
	}
	return pairs
}
